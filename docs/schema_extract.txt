-- =====================================================================
-- База данных ИС фитнес-клуба (учебная работа, группа 26-11С)
-- СУБД: PostgreSQL 14+ (стандартный SQL, идентификаторы на английском,
-- комментарии на русском). Источник: реестр требований из презентации -- (US-01..US-06, FR-01..FR-06, BR-01, CON-01, NFR-01..NFR-05, UC-01).
-- =====================================================================
-- ---------- 1. Справочники и пользователи (FR-01, NFR-02) ----------
CREATE TABLE role (
    role_id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,     code        VARCHAR(20)  NOT NULL UNIQUE,          -- client / trainer / admin
    title       VARCHAR(60)  NOT NULL
);
CREATE TABLE app_user (                                 -- учётная запись 
(регистрация и вход)
    user_id        INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    role_id        INTEGER      NOT NULL REFERENCES role(role_id),     email          VARCHAR(120) NOT NULL UNIQUE,        -- US-01: email не должен повторяться
    password_hash  VARCHAR(255) NOT NULL,               -- пароль хранится только в виде хеша
    full_name      VARCHAR(120) NOT NULL,
    phone          VARCHAR(20),
    is_active      BOOLEAN      NOT NULL DEFAULT TRUE,     created_at     TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP );
CREATE TABLE client (                                   -- профиль клиента
    client_id         INTEGER PRIMARY KEY REFERENCES app_user(user_id),
    birth_date        DATE,     pd_consent        BOOLEAN   NOT NULL DEFAULT FALSE, -- CON-02: согласие на обработку ПДн
    pd_consent_date   TIMESTAMP
);
CREATE TABLE trainer (                                  -- профиль тренера
    trainer_id      INTEGER PRIMARY KEY REFERENCES app_user(user_id),
    specialization  VARCHAR(120)
);

CREATE TABLE club_rule (                                -- настраиваемые правила клуба (BR-01, US-05)
    rule_code    VARCHAR(40) PRIMARY KEY,
    rule_value   VARCHAR(40) NOT NULL,
    description  VARCHAR(255) NOT NULL,
    is_agreed    BOOLEAN NOT NULL DEFAULT FALSE         -- значение согласовано с заказчиком? );
-- ---------- 2. Тарифы, оплата, абонементы (FR-02, FR-03, NFR-03) ---------CREATE TABLE tariff (
    tariff_id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name          VARCHAR(80)   NOT NULL UNIQUE,
    price         NUMERIC(10,2) NOT NULL CHECK (price >= 0),
    duration_days INTEGER       NOT NULL CHECK (duration_days > 0),     visits_limit  INTEGER       CHECK (visits_limit > 0),   -- NULL = безлимит
    is_published  BOOLEAN       NOT NULL DEFAULT TRUE       -- UC-01: тарифы опубликованы
);
CREATE TABLE payment (                                  -- UC-01: платёж через провайдера
    payment_id      INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    client_id       INTEGER       NOT NULL REFERENCES client(client_id),
    tariff_id       INTEGER       NOT NULL REFERENCES tariff(tariff_id),
    amount          NUMERIC(10,2) NOT NULL CHECK (amount >= 0),
    status          VARCHAR(12)   NOT NULL DEFAULT 'pending'
                    CHECK (status IN ('pending','succeeded','failed')),     provider_txn_id VARCHAR(80),                         -- номер операции у провайдера
    created_at      TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    paid_at         TIMESTAMP,     UNIQUE (payment_id, status)                          -- нужно для составного ключа ниже
);
CREATE TABLE membership (                               -- абонемент клиента
    membership_id  INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    client_id      INTEGER     NOT NULL REFERENCES client(client_id),
    tariff_id      INTEGER     NOT NULL REFERENCES tariff(tariff_id),
    payment_id     INTEGER     NOT NULL UNIQUE,
    payment_status VARCHAR(12) NOT NULL DEFAULT 'succeeded' CHECK 
(payment_status = 'succeeded'),
    start_date     DATE        NOT NULL,
    end_date       DATE        NOT NULL,
    visits_left    INTEGER     CHECK (visits_left >= 0),   -- NULL = безлимит
    status         VARCHAR(10) NOT NULL DEFAULT 'active'                    CHECK (status IN ('active','frozen','expired')),
    CHECK (end_date >= start_date),
    -- NFR-03: абонемент возможен только для платежа со статусом succeeded     FOREIGN KEY (payment_id, payment_status) REFERENCES payment(payment_id, status)
);
-- ---------- 3. Расписание и записи (FR-04, FR-05, FR-06, BR-01) ---------CREATE TABLE hall (
    hall_id   INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name      VARCHAR(60) NOT NULL UNIQUE,
    capacity  INTEGER     NOT NULL CHECK (capacity > 0) );
CREATE TABLE class_type (                               -- вид занятия: йога, пилатес, кроссфит...
    class_type_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name          VARCHAR(80) NOT NULL UNIQUE,
    description   VARCHAR(255)
);
CREATE TABLE class_session (                            -- конкретное занятие в расписании
    session_id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    class_type_id  INTEGER     NOT NULL REFERENCES class_type(class_type_id),
    trainer_id     INTEGER     NOT NULL REFERENCES trainer(trainer_id),
    hall_id        INTEGER     NOT NULL REFERENCES hall(hall_id),
    starts_at      TIMESTAMP   NOT NULL,
    ends_at        TIMESTAMP   NOT NULL,
    seats_total    INTEGER     NOT NULL CHECK (seats_total > 0),
    status         VARCHAR(10) NOT NULL DEFAULT 'scheduled'                    CHECK (status IN ('scheduled','cancelled','done')),
    CHECK (ends_at > starts_at)
);
CREATE TABLE booking (                                  -- запись клиента на занятие
    booking_id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    session_id     INTEGER     NOT NULL REFERENCES class_session(session_id),
    client_id      INTEGER     NOT NULL REFERENCES client(client_id),
    membership_id  INTEGER     NOT NULL REFERENCES membership(membership_id),
    status         VARCHAR(10) NOT NULL DEFAULT 'booked'
                   CHECK (status IN 
('booked','cancelled','attended','no_show')),
    created_at     TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    cancelled_at   TIMESTAMP
);
-- Один клиент не может иметь две активные записи на одно занятие
CREATE UNIQUE INDEX ux_booking_active ON booking(session_id, client_id)
    WHERE status IN ('booked','attended');
CREATE TABLE attendance (                               -- отметка посещения (US-06, UC-04)
    attendance_id  INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    booking_id     INTEGER     NOT NULL UNIQUE REFERENCES booking(booking_id),
    marked_by      INTEGER     NOT NULL REFERENCES trainer(trainer_id),
    marked_at      TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
    result         VARCHAR(10) NOT NULL CHECK (result IN 
('attended','no_show'))
);
-- ---------- 4. Реестр требований и трассируемость (ISO 29148) ----------
CREATE TABLE user_story (
    story_code  VARCHAR(10) PRIMARY KEY,                -- US-01
    actor_role  VARCHAR(60) NOT NULL,                   -- "Как ..."
    action      VARCHAR(255) NOT NULL,                  -- "я хочу ..."
    value_text  VARCHAR(255) NOT NULL,                  -- "чтобы ..."
    moscow      VARCHAR(6) NOT NULL CHECK (moscow IN 
('Must','Should','Could','Won''t'))
);
CREATE TABLE requirement (
    req_code    VARCHAR(10) PRIMARY KEY,                -- FR-01, NFR-01, BR01, CON-01
    req_class   VARCHAR(20) NOT NULL                 CHECK (req_class IN 
('functional','nonfunctional','business_rule','constraint')),
    statement   VARCHAR(500) NOT NULL,
    priority    VARCHAR(6) NOT NULL CHECK (priority IN 
('Must','Should','Could','Won''t')),
    status      VARCHAR(30) NOT NULL DEFAULT 'draft',
    story_code  VARCHAR(10) REFERENCES user_story(story_code)  -- трассируемость к источнику );
CREATE TABLE acceptance_criterion (                     -- "Дано - Когда - 
Тогда"
    criterion_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    req_code     VARCHAR(10) NOT NULL REFERENCES requirement(req_code),
    given_text   VARCHAR(255) NOT NULL,
    when_text    VARCHAR(255),
    then_text    VARCHAR(255) NOT NULL
);
-- ---------- 5. Индексы и представления ----------
CREATE INDEX ix_session_starts   ON class_session(starts_at);
CREATE INDEX ix_booking_client   ON booking(client_id); CREATE INDEX ix_membership_client ON membership(client_id, status);
-- Свободные места на занятии (BR-01, FR-05)
CREATE VIEW v_session_free_seats AS
SELECT s.session_id, s.starts_at, ct.name AS class_name,        s.seats_total,
       s.seats_total - COUNT(b.booking_id) AS seats_free FROM class_session s
JOIN class_type ct ON ct.class_type_id = s.class_type_id LEFT JOIN booking b ON b.session_id = s.session_id AND b.status IN 
('booked','attended')
WHERE s.status = 'scheduled'
GROUP BY s.session_id, s.starts_at, ct.name, s.seats_total;
-- История посещений клиента (US-06, FR-02)
CREATE VIEW v_client_history AS
SELECT b.client_id, s.starts_at, ct.name AS class_name, b.status
FROM booking b
JOIN class_session s ON s.session_id = b.session_id JOIN class_type ct ON ct.class_type_id = s.class_type_id;
